//= Production-ready auction system
//===== Features: ============================================
//= - RODEX delivery, 6% tax, Anti-Dupe (C++ side)
//= - Simplify — listing only. Search via client window.

-	script	MarketBoard	858,{
OnClick:
	mes "[Market Board]";
	mes "Welcome to the Global Market!";
	mes " ";
	mes "How can I help you today?";
	next;

	switch(select("Buy (Market Search):Sell (Vending UI):Cancel")) {
		case 1:
			atcommand "@market buy";
			break;
		case 2:
			atcommand "@market sell";
			break;
		case 3:
			close;
	}
	close;
}

prontera,150,150,5	duplicate(MarketBoard)	Market Board#prt	858
alberta,150,150,5	duplicate(MarketBoard)	Market Board#alb	858
morocc,150,150,5	duplicate(MarketBoard)	Market Board#moc	858
